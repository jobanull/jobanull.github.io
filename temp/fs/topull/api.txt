// ==================== FUNGSI HELPER ====================

function sendResponse($data, $status = 200) {
    http_response_code($status);
    echo json_encode($data);
    exit;
}

function formatData($row, $type = 'parent') {
    // Ambil opt dari hasil join jika ada
    $optField = ($type === 'parent') ? 'opt' : 'opt';
    
    // Cek apakah ada field opt_code dari view atau hasil join
    $opt = isset($row['opt']) ? $row['opt'] : 
           (isset($row['opt_code']) ? $row['opt_code'] : 
           (isset($row['parent_opt']) ? $row['parent_opt'] : 'PARENT'));
    
    if ($type === 'parent') {
        return [
            'id' => (int)$row['id'],
            'location' => $row['location'],
            'startTime' => $row['start_time'] ? date('h:i:s', strtotime($row['start_time'])) : '',
            'startAmPm' => $row['start_ampm'],
            'startDate' => $row['start_date'] ? date('m/d/Y', strtotime($row['start_date'])) : '',
            'stopTime' => $row['stop_time'] ? date('h:i:s', strtotime($row['stop_time'])) : '',
            'stopAmPm' => $row['stop_ampm'] ?? '',
            'stopDate' => $row['stop_date'] ? date('m/d/Y', strtotime($row['stop_date'])) : '',
            'status' => $row['status'],
            'opt' => $opt,
            'vdc' => $row['vdc']
        ];
    } else {
        return [
            'id' => (int)$row['id'],
            'parent_id' => (int)$row['parent_id'],
            'location' => $row['location'],
            'startTime' => $row['start_time'] ? date('h:i:s', strtotime($row['start_time'])) : '',
            'startAmPm' => $row['start_ampm'],
            'startDate' => $row['start_date'] ? date('m/d/Y', strtotime($row['start_date'])) : '',
            'stopTime' => $row['stop_time'] ? date('h:i:s', strtotime($row['stop_time'])) : '',
            'stopAmPm' => $row['stop_ampm'] ?? '',
            'stopDate' => $row['stop_date'] ? date('m/d/Y', strtotime($row['stop_date'])) : '',
            'status' => $row['status'],
            'opt' => $opt,
            'vdc' => $row['vdc']
        ];
    }
}

function logAudit($tableName, $recordId, $action, $oldData = null, $newData = null) {
    try {
        $pdo = getDBConnection();
        $stmt = $pdo->prepare("
            INSERT INTO prtg_audit_log 
            (table_name, record_id, action, old_data, new_data, ip_address, user_agent) 
            VALUES (?, ?, ?, ?, ?, ?, ?)
        ");
        
        $ip = $_SERVER['REMOTE_ADDR'] ?? null;
        $userAgent = $_SERVER['HTTP_USER_AGENT'] ?? null;
        
        $stmt->execute([
            $tableName,
            $recordId,
            $action,
            $oldData ? json_encode($oldData) : null,
            $newData ? json_encode($newData) : null,
            $ip,
            $userAgent
        ]);
    } catch (Exception $e) {
        error_log("Audit log failed: " . $e->getMessage());
    }
}

// ==================== FUNGSI UPPERCASE ====================

function toUpperCase($value) {
    if (is_string($value)) {
        return strtoupper(trim($value));
    }
    return $value;
}

function sanitizeAndUpperCase($data) {
    $uppercased = [];
    foreach ($data as $key => $value) {
        if (is_string($value)) {
            $uppercased[$key] = strtoupper(trim($value));
        } else {
            $uppercased[$key] = $value;
        }
    }
    return $uppercased;
}

                $input = json_decode(file_get_contents('php://input'), true);
                if (!$input) {
                    sendResponse(['success' => false, 'message' => 'Invalid JSON'], 400);
                }
                
                // ============= UPPERCASE LOCATION =============
                $location = strtoupper(trim($input['location'] ?? ''));
                if (empty($location)) {
                    sendResponse(['success' => false, 'message' => 'Lokasi tidak boleh kosong'], 400);
                }
                
                $now = new DateTime();
                $stopTime = $now->format('H:i:s');
                $stopAmPm = strtoupper($now->format('A'));
                $stopDate = $now->format('Y-m-d');
                
                $stmt = $pdo->prepare("
                    UPDATE prtg_parent p
                    JOIN prtg_options o ON o.id = p.opt_id
                    SET p.stop_time = ?, p.stop_ampm = ?, p.stop_date = ?, p.status = 'paused' 
                    WHERE p.location = ? AND p.status = 'active' AND o.code = 'PARENT'
                ");
                $stmt->execute([$stopTime, $stopAmPm, $stopDate, $location]);
                $updatedCount = $stmt->rowCount();
                
                if ($updatedCount == 0) {
                    sendResponse([
                        'success' => false, 
                        'message' => "Tidak ada data PARENT active untuk \"$location\""
                    ]);
                }
                
                logAudit('prtg_parent', 0, 'UPDATE');
                sendResponse([
                    'success' => true,
                    'message' => "UP BY GENSET",
                    'updated_count' => $updatedCount
                ]);
            }
            
            // PUT UNPAUSE BY LOCATION (hanya untuk PARENT)
            elseif ($path === 'prtg/unpause-by-location') {
                $input = json_decode(file_get_contents('php://input'), true);
                if (!$input) {
                    sendResponse(['success' => false, 'message' => 'Invalid JSON'], 400);
                }
                
                // ============= UPPERCASE LOCATION =============
                $location = strtoupper(trim($input['location'] ?? ''));
                if (empty($location)) {
                    sendResponse(['success' => false, 'message' => 'Lokasi tidak boleh kosong'], 400);
                }
                
                $stmt = $pdo->prepare("
                    UPDATE prtg_parent p
                    JOIN prtg_options o ON o.id = p.opt_id
                    SET p.stop_time = NULL, p.stop_ampm = NULL, p.stop_date = NULL, p.status = 'active' 
                    WHERE p.location = ? AND p.status = 'paused' AND o.code = 'PARENT'
                ");
                $stmt->execute([$location]);
                $updatedCount = $stmt->rowCount();
                
                if ($updatedCount == 0) {
                    sendResponse([
                        'success' => false, 
                        'message' => "Tidak ada data PARENT paused untuk \"$location\""
                    ]);
                }
                
                logAudit('prtg_parent', 0, 'UPDATE');
                sendResponse([
                    'success' => true,
                    'message' => "UP BY PLN",
                    'updated_count' => $updatedCount
                ]);
            }
            
            // ==================== NEW: PUT REOPEN BY LOCATION ====================
            elseif ($path === 'prtg/reopen-by-location') {
                $input = json_decode(file_get_contents('php://input'), true);
                if (!$input) {
                    sendResponse(['success' => false, 'message' => 'Invalid JSON'], 400);
                }
                
                // UPPERCASE LOCATION
                $location = strtoupper(trim($input['location'] ?? ''));
                if (empty($location)) {
                    sendResponse(['success' => false, 'message' => 'Lokasi tidak boleh kosong'], 400);
                }
                
                // Cari PARENT dengan status done
                $stmt = $pdo->prepare("
                    SELECT p.id, p.start_time, p.start_date, p.start_ampm, p.vdc, o.code as opt
                    FROM prtg_parent p
                    JOIN prtg_options o ON o.id = p.opt_id
                    WHERE p.location = ? AND p.status = 'done' AND o.code = 'PARENT'
                    ORDER BY p.id DESC LIMIT 1
                ");
                $stmt->execute([$location]);
                $parent = $stmt->fetch();
                
                if (!$parent) {
                    sendResponse([
                        'success' => false, 
                        'message' => "Tidak ada data PARENT dengan status DONE untuk \"$location\""
                    ]);
                }
                
                $parentId = $parent['id'];
                $oldParentData = $parent;
                
                // Ambil data CHILD yang terkait sebelum update
                $stmt = $pdo->prepare("
                    SELECT c.*, o.code as opt 
                    FROM prtg_child c
                    JOIN prtg_options o ON o.id = c.opt_id
                    WHERE c.parent_id = ? AND c.status = 'done'
                ");
                $stmt->execute([$parentId]);
                $oldChildData = $stmt->fetchAll();
                
                // Update PARENT: status menjadi active, hapus stop_time, stop_ampm, stop_date
                $stmt = $pdo->prepare("
                    UPDATE prtg_parent 
                    SET status = 'active', stop_time = NULL, stop_ampm = NULL, stop_date = NULL 
                    WHERE id = ?
                ");
                $stmt->execute([$parentId]);
                $parentUpdated = $stmt->rowCount();
                
                // Update CHILD yang terkait (status done -> active, hapus stop data)
                $stmt = $pdo->prepare("
                    UPDATE prtg_child 
                    SET status = 'active', stop_time = NULL, stop_ampm = NULL, stop_date = NULL 
                    WHERE parent_id = ? AND status = 'done'
                ");
                $stmt->execute([$parentId]);
                $childUpdated = $stmt->rowCount();
                
                if ($parentUpdated == 0 && $childUpdated == 0) {
                    sendResponse(['success' => false, 'message' => 'Tidak ada data yang di-reopen']);
                }
                
                // Log audit
                logAudit('prtg_parent', $parentId, 'REOPEN', $oldParentData);
                if (!empty($oldChildData)) {
                    logAudit('prtg_child', $parentId, 'REOPEN', $oldChildData);
                }
                
                // Ambil data terbaru untuk response
                $stmt = $pdo->prepare("
                    SELECT p.*, o.code as opt 
                    FROM prtg_parent p
                    JOIN prtg_options o ON o.id = p.opt_id
                    WHERE p.id = ?
                ");
                $stmt->execute([$parentId]);
                $newParentData = $stmt->fetch();
                
                sendResponse([
                    'success' => true,
                    'message' => "✅ \"$location\" Berhasil direopen",
                    'parent_updated' => $parentUpdated,
                    'child_updated' => $childUpdated,
                    'data' => $newParentData ? formatData($newParentData, 'parent') : null
                ]);
            }
            
            else {
                sendResponse(['success' => false, 'message' => 'Endpoint not found: ' . $path], 404);
            }
            break;

        // ==================== DELETE ====================
        case 'DELETE':
            // Parse path
            $pathParts = explode('/', $path);
            $endpoint = $pathParts[0] ?? '';
            $subEndpoint = $pathParts[1] ?? '';
            $id = $pathParts[2] ?? null;
            
            // DELETE BY ID (support parent & child)
            if ($endpoint === 'prtg' && is_numeric($subEndpoint)) {
                $id = (int)$subEndpoint;
                
                // Cek di parent dulu
                $stmt = $pdo->prepare("SELECT * FROM prtg_parent WHERE id = ?");
                $stmt->execute([$id]);
                $item = $stmt->fetch();
                
                if ($item) {
                    // Delete child dulu
                    $pdo->prepare("DELETE FROM prtg_child WHERE parent_id = ?")->execute([$id]);
                    // Delete parent
                    $stmt = $pdo->prepare("DELETE FROM prtg_parent WHERE id = ?");
                    $stmt->execute([$id]);
                    
                    logAudit('prtg_parent', $id, 'DELETE', $item);
                    
                    sendResponse([
                        'success' => true,
                        'message' => "✅ DATA TERHAPUS!"
                    ]);
                } else {
                    // Cek di child
                    $stmt = $pdo->prepare("SELECT * FROM prtg_child WHERE id = ?");
                    $stmt->execute([$id]);
                    $item = $stmt->fetch();
                    
                    if ($item) {
                        $stmt = $pdo->prepare("DELETE FROM prtg_child WHERE id = ?");
                        $stmt->execute([$id]);
                        
                        logAudit('prtg_child', $id, 'DELETE', $item);
                        
                        sendResponse([
                            'success' => true,
                            'message' => "DELETED: \"{$item['location']}\" deleted!"
                        ]);
                    } else {
                        sendResponse(['success' => false, 'message' => 'Data tidak ditemukan'], 404);
                    }
                }
            }
            
            // DELETE BY LOCATION OPT
            elseif ($path === 'prtg/delete-by-location-opt') {
                $input = json_decode(file_get_contents('php://input'), true);
                if (!$input) {
                    sendResponse(['success' => false, 'message' => 'Invalid JSON'], 400);
                }
                
                // ============= UPPERCASE INPUT =============
                $input = sanitizeAndUpperCase($input);
                $location = $input['location'] ?? '';
                $optCode = $input['opt'] ?? '';
                
                if (empty($location) || empty($optCode)) {
                    sendResponse(['success' => false, 'message' => 'Location and opt required'], 400);
                }
                
                $table = ($optCode === 'CHILD') ? 'prtg_child' : 'prtg_parent';
                
                // Ambil data sebelum delete untuk audit
                $stmt = $pdo->prepare("
                    SELECT t.* FROM $table t
                    JOIN prtg_options o ON o.id = t.opt_id
                    WHERE t.location = ? AND o.code = ?
                ");
                $stmt->execute([$location, $optCode]);
                $oldData = $stmt->fetchAll();
                
                if ($optCode === 'CHILD') {
                    $stmt = $pdo->prepare("
                        DELETE c FROM prtg_child c
                        JOIN prtg_options o ON o.id = c.opt_id
                        WHERE c.location = ? AND o.code = ?
                    ");
                } else {
                    $stmt = $pdo->prepare("
                        DELETE p FROM prtg_parent p
                        JOIN prtg_options o ON o.id = p.opt_id
                        WHERE p.location = ? AND o.code = ?
                    ");
                }
                $stmt->execute([$location, $optCode]);
                $deletedCount = $stmt->rowCount();
                
                if ($deletedCount == 0) {
                    sendResponse(['success' => false, 'message' => 'Data not found'], 404);
                }
                
                logAudit($table, 0, 'DELETE', $oldData);
                sendResponse([
                    'success' => true,
                    'message' => "Deleted $deletedCount items",
                    'deleted_count' => $deletedCount
                ]);
            }
            
            // DELETE BY LOCATION DATE OPT
            elseif ($path === 'prtg/delete-by-location-date-opt') {
                $input = json_decode(file_get_contents('php://input'), true);
                if (!$input) {
                    sendResponse(['success' => false, 'message' => 'Invalid JSON'], 400);
                }
                
                // UPPERCASE INPUT
                $input = sanitizeAndUpperCase($input);
                $location = $input['location'] ?? '';
                $startDate = $input['startDate'] ?? '';
                $optCode = $input['opt'] ?? '';
                
                if (empty($location) || empty($startDate) || empty($optCode)) {
                    sendResponse(['success' => false, 'message' => 'Location, startDate, and opt required'], 400);
                }
                
                // Convert startDate dari MM/DD/YYYY ke YYYY-MM-DD
                $dateObj = DateTime::createFromFormat('m/d/Y', $startDate);
                if (!$dateObj) {
                    sendResponse(['success' => false, 'message' => 'Invalid date format. Use MM/DD/YYYY'], 400);
                }
                $startDateFormatted = $dateObj->format('Y-m-d');
                
                $table = ($optCode === 'CHILD') ? 'prtg_child' : 'prtg_parent';
                
                // Ambil data sebelum delete untuk audit
                $stmt = $pdo->prepare("
                    SELECT t.* FROM $table t
                    JOIN prtg_options o ON o.id = t.opt_id
                    WHERE t.location = ? AND t.start_date = ? AND o.code = ?
                ");
                $stmt->execute([$location, $startDateFormatted, $optCode]);
                $oldData = $stmt->fetchAll();
                
                if ($optCode === 'CHILD') {
                    $stmt = $pdo->prepare("
                        DELETE c FROM prtg_child c
                        JOIN prtg_options o ON o.id = c.opt_id
                        WHERE c.location = ? AND c.start_date = ? AND o.code = ?
                    ");
                } else {
                    $stmt = $pdo->prepare("
                        DELETE p FROM prtg_parent p
                        JOIN prtg_options o ON o.id = p.opt_id
                        WHERE p.location = ? AND p.start_date = ? AND o.code = ?
                    ");
                }
                $stmt->execute([$location, $startDateFormatted, $optCode]);
                $deletedCount = $stmt->rowCount();
                
                if ($deletedCount == 0) {
                    sendResponse(['success' => false, 'message' => 'Data not found'], 404);
                }
                
                logAudit($table, 0, 'DELETE', $oldData);
                sendResponse([
                    'success' => true,
                    'message' => "Deleted $deletedCount items",
                    'deleted_count' => $deletedCount
                ]);
            }
            
            else {
                sendResponse(['success' => false, 'message' => 'Endpoint not found: ' . $path], 404);
            }
            break;

        // ==================== METHOD NOT ALLOWED ====================
        default:
            sendResponse(['success' => false, 'message' => 'Method not allowed'], 405);
    }
} catch (Exception $e) {
    http_response_code(500);
    echo json_encode([
        'success' => false, 
        'message' => 'Server error: ' . $e->getMessage()
    ]);
    error_log("API Error: " . $e->getMessage() . "\n" . $e->getTraceAsString());
}
