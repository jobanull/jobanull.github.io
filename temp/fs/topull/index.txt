    <!-- Tambahkan script untuk override loadContent dan initialize PRTG ulang -->
    <script>
    (function() {
        'use strict';
        
        // Simpan referensi fungsi asli
        const originalLoadContent = window.loadContent;
        
        // Override loadContent
        window.loadContent = function(target) {
            // Jika target adalah PRTG, re-initialize
            if (target === 'prtg') {
                console.log('🔄 Loading PRTG content...');
                
                // Panggil fungsi initializePrtg jika ada
                if (typeof initializePrtg === 'function') {
                    // Reset flag initialized agar bisa diinisialisasi ulang
                    window._prtgInitialized = false;
                    
                    // Kosongkan content terlebih dahulu
                    const content = document.getElementById('content');
                    if (content) {
                        content.innerHTML = '<div id="loading">Loading PRTG...</div>';
                    }
                    
                    // Inisialisasi PRTG dengan delay kecil
                    setTimeout(function() {
                        initializePrtg();
                        console.log('✅ PRTG re-initialized');
                    }, 100);
                } else {
                    console.warn('⚠️ initializePrtg function not found');
                    // Fallback: panggil fungsi asli
                    if (originalLoadContent) {
                        originalLoadContent(target);
                    }
                }
                return;
            }
            
            // Untuk target lain, gunakan fungsi asli
            if (originalLoadContent) {
                originalLoadContent(target);
            }
        };
        
        // Tambahkan event listener untuk semua tombol PRTG
        document.addEventListener('DOMContentLoaded', function() {
            // Cari semua tombol dengan data-target="prtg"
            const prtgButtons = document.querySelectorAll('button[data-target="prtg"]');
            prtgButtons.forEach(function(btn) {
                // Tambahkan event listener klik
                btn.addEventListener('click', function(e) {
                    // Reset flag PRTG
                    window._prtgInitialized = false;
                    
                    // Hapus class active dari semua tombol
                    document.querySelectorAll('.nav-list button').forEach(function(b) {
                        b.classList.remove('active');
                    });
                    
                    // Tambahkan class active ke tombol yang diklik
                    this.classList.add('active');
                });
            });
        });
    })();
    </script>
