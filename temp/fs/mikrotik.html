MIKROTIK MPLS, & ISIS

# ISIS & MPLS CONFIGURATION
# TOPOLOGY
# R1 <> R2 <> R3

# ==========================================
# R1
# ==========================================

# ================================
# SYSTEM IDENTITY
# ================================
/system identity
set name=R1

# ================================
# LOOPBACK CONFIGURATION
# ================================
/interface bridge
add name=lo1

/ip address
add address=1.1.1.1/32 interface=lo1

# ================================
# CORE INTERFACE CONFIGURATION
# ================================
/ip address
add address=10.10.12.1/30 interface=ether1

# ================================
# ROUTER-ID CONFIGURATION
# ================================
/routing id
add id=1.1.1.1 name=RID

# ================================
# ISIS CONFIGURATION
# ================================
/routing isis instance
add name=CORE afi=ip areas=49.0001 system-id=0000.0000.0001

/routing isis interface-template
add interfaces=ether1 instance=CORE levels=l2
add interfaces=lo1 instance=CORE passive levels=l2

# ================================
# MPLS CONFIGURATION
# ================================
/mpls interface
add interface=ether1

/mpls ldp
add lsr-id=1.1.1.1 transport-addresses=1.1.1.1

/mpls ldp interface
add interface=ether1
